cmake_minimum_required(VERSION 3.22)
project(HikaruNativeBoot LANGUAGES C CXX)
add_subdirectory(../../core/hw/hikaru "${CMAKE_CURRENT_BINARY_DIR}/hikaru")
add_executable(hikaru_boot boot.cpp)
target_link_libraries(hikaru_boot PRIVATE hollycast_hikaru_bus)
if(MSVC)
    target_compile_options(hikaru_boot PRIVATE /W4)
else()
    target_compile_options(hikaru_boot PRIVATE -Wall -Wextra -Werror)
endif()

option(BUILD_HIKARU_PLAYER "Build the experimental desktop OpenGL Air Trix player" ON)
if(BUILD_HIKARU_PLAYER)
    find_package(SDL2 2.0.18 REQUIRED)
    add_subdirectory(../../core/hw/hikaru/valkyrie "${CMAKE_CURRENT_BINARY_DIR}/valkyrie")
    add_executable(hollycast_hikaru play.cpp)
    target_link_libraries(hollycast_hikaru PRIVATE hollycast_hikaru_bus hollycast_hikaru_valkyrie SDL2::SDL2)
    if(APPLE)
        target_compile_definitions(hollycast_hikaru PRIVATE GL_SILENCE_DEPRECATION)
    else()
        target_sources(hollycast_hikaru PRIVATE ../../core/deps/glad/src/gl.c)
        target_include_directories(hollycast_hikaru PRIVATE ../../core/deps/glad/include)
        target_link_libraries(hollycast_hikaru PRIVATE ${CMAKE_DL_LIBS})
    endif()
endif()
