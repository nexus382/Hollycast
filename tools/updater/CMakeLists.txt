add_executable(HollycastUpdater WIN32 windows_updater.cpp)
target_compile_features(HollycastUpdater PRIVATE cxx_std_17)
target_compile_definitions(HollycastUpdater PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(HollycastUpdater PRIVATE "${PROJECT_SOURCE_DIR}/core/deps/json")
target_link_libraries(HollycastUpdater PRIVATE winhttp bcrypt shell32)
if(TARGET libzip::zip)
    target_link_libraries(HollycastUpdater PRIVATE libzip::zip)
else()
    target_link_libraries(HollycastUpdater PRIVATE PkgConfig::LIBZIP)
endif()
if(MINGW)
    target_link_options(HollycastUpdater PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
add_dependencies(${PROJECT_NAME} HollycastUpdater)
install(TARGETS HollycastUpdater RUNTIME DESTINATION bin)
